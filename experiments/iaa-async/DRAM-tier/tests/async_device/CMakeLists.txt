cmake_minimum_required(VERSION 3.20)
project(zipcache_device_surrogate C)
enable_testing()
set(CMAKE_C_STANDARD 11)
option(SD_ASAN "ASan/UBSan" OFF)
option(SD_TSAN "TSan" OFF)
if(SD_ASAN AND SD_TSAN)
  message(FATAL_ERROR "Separate sanitizer builds required")
endif()
if(SD_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(SD_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LZ4 REQUIRED IMPORTED_TARGET liblz4)
find_package(ZLIB REQUIRED) # framing CRC only; synthetic routes never call a codec
set(LIB ../../lib)
add_library(device STATIC ${LIB}/async_copy_device.c)
target_include_directories(device PUBLIC ${LIB})
target_link_libraries(device PUBLIC Threads::Threads)
add_executable(device_check device_check.c)
target_link_libraries(device_check device)
add_test(NAME device_contract COMMAND device_check)
set_tests_properties(device_contract PROPERTIES TIMEOUT 30)
foreach(layout 2 3)
  add_library(tree_l${layout} STATIC ${LIB}/bplustree.c ${LIB}/bplustree_compressed.c)
  target_compile_definitions(tree_l${layout} PUBLIC ZIPCACHE_DEVICE_SURROGATE=1 ZIPCACHE_AGG_EXPERIMENT=1 ZIPCACHE_AGG_LAYOUT=${layout} ZIPCACHE_DIRECT_REBUILD=1 HAVE_ZLIB=1)
  target_include_directories(tree_l${layout} PUBLIC ${LIB} ../btree)
  target_link_libraries(tree_l${layout} PUBLIC device PkgConfig::LZ4 ZLIB::ZLIB Threads::Threads ${CMAKE_DL_LIBS})
  add_executable(tree_check_l${layout} tree_check.c)
  target_link_libraries(tree_check_l${layout} tree_l${layout})
  add_test(NAME tree_contract_l${layout} COMMAND tree_check_l${layout})
  set_tests_properties(tree_contract_l${layout} PROPERTIES TIMEOUT 120 ENVIRONMENT "BTREE_MEMORY_AUDIT=1")
  add_executable(device_bench_l${layout} bench.c)
  target_link_libraries(device_bench_l${layout} tree_l${layout})
  add_executable(cross_check_l${layout} cross_check.c)
  target_link_libraries(cross_check_l${layout} tree_l${layout})
  add_test(NAME cross_contract_l${layout} COMMAND cross_check_l${layout})
  set_tests_properties(cross_contract_l${layout} PROPERTIES TIMEOUT 60 ENVIRONMENT "BTREE_MEMORY_AUDIT=1;SD_DIAGNOSTICS=1;SD_TRACE_CAPACITY=8192")
  add_test(NAME cross_tree_l${layout} COMMAND tree_check_l${layout})
  set_tests_properties(cross_tree_l${layout} PROPERTIES TIMEOUT 120 ENVIRONMENT "BTREE_CROSS_LEAF=1;BTREE_CROSS_OUTSTANDING=4;BTREE_AGG_BACKPRESSURE=1;BTREE_MEMORY_AUDIT=1")
  add_test(NAME cross_concurrent_l${layout} COMMAND device_bench_l${layout} async 0.2 4 512 80 0 0)
  set_tests_properties(cross_concurrent_l${layout} PROPERTIES TIMEOUT 60 ENVIRONMENT "BTREE_CROSS_LEAF=1;BTREE_CROSS_OUTSTANDING=4;BTREE_AGG_BACKPRESSURE=1;SD_QUEUE=1;SD_BYTES=65536")
  add_test(NAME tree_concurrent_l${layout} COMMAND device_bench_l${layout} async 0.1 4 512 80 0 0)
  set_tests_properties(tree_concurrent_l${layout} PROPERTIES TIMEOUT 60)
  add_test(NAME cross_full_workspace_l${layout} COMMAND cross_check_l${layout})
  set_tests_properties(cross_full_workspace_l${layout} PROPERTIES TIMEOUT 60 ENVIRONMENT "BTREE_CROSS_COMPACT_WORKSPACE=0;BTREE_MEMORY_AUDIT=1")
  add_test(NAME cross_tail_pressure_l${layout} COMMAND device_bench_l${layout} async 0.2 4 512 80 0 0)
  set_tests_properties(cross_tail_pressure_l${layout} PROPERTIES TIMEOUT 60 ENVIRONMENT "BTREE_CROSS_LEAF=1;BTREE_AGG_BACKPRESSURE=1;SD_QUEUE=1;SD_BYTES=65536;SD_TAIL_EVERY=5;SD_TAIL_MULTIPLIER=5;SD_BUSY_NEXT=3;SD_DIAGNOSTICS=1;SD_TRACE_CAPACITY=8192")
  add_test(NAME cross_sequential_l${layout} COMMAND device_bench_l${layout} async 0.2 4 512 80 0 0)
  set_tests_properties(cross_sequential_l${layout} PROPERTIES TIMEOUT 60 ENVIRONMENT "BTREE_CROSS_LEAF=1;BTREE_AGG_BACKPRESSURE=1;BTREE_TRACE_ORDER=sequential")
endforeach()
foreach(layout 2 3)
  add_executable(reuse_check_l${layout} reuse_check.c)
  target_link_libraries(reuse_check_l${layout} tree_l${layout})
  foreach(codec async lz4 zlib)
    add_test(NAME reuse_${codec}_l${layout} COMMAND reuse_check_l${layout} ${codec})
    set_tests_properties(reuse_${codec}_l${layout} PROPERTIES TIMEOUT 60 ENVIRONMENT "BTREE_MEMORY_AUDIT=1")
  endforeach()
endforeach()
# Exercise the existing structural/failure/concurrency contracts with reuse on.
add_test(NAME reuse_structural COMMAND tree_check_l3)
set_tests_properties(reuse_structural PROPERTIES TIMEOUT 120 ENVIRONMENT "BTREE_MEMORY_AUDIT=1;BTREE_REUSE_UNCHANGED_BLOCKS=1;BTREE_CROSS_LEAF=1;BTREE_AGG_BACKPRESSURE=1")
add_test(NAME reuse_cross_contract COMMAND cross_check_l3)
set_tests_properties(reuse_cross_contract PROPERTIES TIMEOUT 60 ENVIRONMENT "BTREE_MEMORY_AUDIT=1;BTREE_REUSE_UNCHANGED_BLOCKS=1")
add_test(NAME reuse_concurrent COMMAND device_bench_l3 async 0.5 4 512 80 0 0)
set_tests_properties(reuse_concurrent PROPERTIES TIMEOUT 60 ENVIRONMENT "BTREE_REUSE_UNCHANGED_BLOCKS=1;BTREE_CROSS_LEAF=1;BTREE_AGG_BACKPRESSURE=1;SD_QUEUE=1;SD_BYTES=65536")
foreach(codec lz4 zlib)
  add_test(NAME reuse_concurrent_${codec} COMMAND device_bench_l3 ${codec} 0.3 4 2048 80 0 0)
  set_tests_properties(reuse_concurrent_${codec} PROPERTIES TIMEOUT 60 ENVIRONMENT "BTREE_REUSE_UNCHANGED_BLOCKS=1;BTREE_CROSS_LEAF=0;BTREE_AGG_BACKPRESSURE=1")
endforeach()

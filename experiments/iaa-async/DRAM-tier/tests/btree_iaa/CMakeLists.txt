cmake_minimum_required(VERSION 3.20)
project(zipcache_btree_iaa C CXX)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(AGG_ASAN "Address/undefined sanitizers" OFF)
option(AGG_TSAN "Thread sanitizer" OFF)
option(AGG_DIRECT_REBUILD "Direct ordered-record compaction for framed single subpages" ON)
option(AGG_REBUILD_DIAGNOSTICS "Count CPU preparation work, not for performance runs" OFF)
if(AGG_ASAN AND AGG_TSAN)
  message(FATAL_ERROR "ASan and TSan need separate builds")
endif()
if(AGG_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(AGG_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(LZ4 REQUIRED IMPORTED_TARGET liblz4)
pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME split_comparison_math COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/check_split_comparison.py)
option(AGG_DISABLE_QPL "Explicit CPU-only local smoke build" OFF)
if(AGG_DISABLE_QPL)
  set(QPL_LIBRARY "")
  set(QPL_INCLUDE_DIR "")
else()
  find_library(QPL_LIBRARY qpl HINTS $ENV{QPL_ROOT}/lib $ENV{QPL_ROOT}/lib64)
  find_path(QPL_INCLUDE_DIR qpl/qpl.h HINTS $ENV{QPL_ROOT}/include)
endif()
set(TREE_LIB "${CMAKE_CURRENT_LIST_DIR}/../../lib")
foreach(layout RANGE 0 3)
  add_library(agg_tree_l${layout} STATIC ${TREE_LIB}/bplustree.c ${TREE_LIB}/bplustree_compressed.c)
  target_compile_definitions(agg_tree_l${layout} PUBLIC ZIPCACHE_AGG_EXPERIMENT=1 ZIPCACHE_AGG_LAYOUT=${layout} HAVE_ZLIB=1 HAVE_ZSTD=1)
  target_compile_definitions(agg_tree_l${layout} PUBLIC ZIPCACHE_DIRECT_REBUILD=$<BOOL:${AGG_DIRECT_REBUILD}>)
  if(layout GREATER 1)
    target_sources(agg_tree_l${layout} PRIVATE ${TREE_LIB}/async_copy_device.c)
    target_compile_definitions(agg_tree_l${layout} PUBLIC ZIPCACHE_DEVICE_SURROGATE=1)
  endif()
  if(AGG_REBUILD_DIAGNOSTICS)
    target_compile_definitions(agg_tree_l${layout} PUBLIC ZIPCACHE_REBUILD_DIAGNOSTICS=1)
  endif()
  target_include_directories(agg_tree_l${layout} PUBLIC ${TREE_LIB} ../btree)
  target_link_libraries(agg_tree_l${layout} PUBLIC PkgConfig::LZ4 PkgConfig::ZSTD ZLIB::ZLIB Threads::Threads ${CMAKE_DL_LIBS})
  if(QPL_LIBRARY AND QPL_INCLUDE_DIR)
    target_compile_definitions(agg_tree_l${layout} PUBLIC HAVE_QPL=1)
    target_include_directories(agg_tree_l${layout} PUBLIC ${QPL_INCLUDE_DIR})
    target_link_libraries(agg_tree_l${layout} PUBLIC ${QPL_LIBRARY})
    if(QPL_LIBRARY MATCHES "\\.a$")
      # QPL's static archive contains C++; use the selected toolchain's runtime.
      target_link_libraries(agg_tree_l${layout} PUBLIC ${CMAKE_CXX_IMPLICIT_LINK_LIBRARIES})
    endif()
  endif()
  add_executable(agg_check_l${layout} checks.c)
  target_link_libraries(agg_check_l${layout} agg_tree_l${layout})
  add_executable(agg_bench_l${layout} bench.c)
  target_link_libraries(agg_bench_l${layout} agg_tree_l${layout})
  if(layout GREATER 1)
    foreach(codec lz4 zlib)
      add_test(NAME l${layout}_cross_${codec} COMMAND agg_check_l${layout} ${codec})
      set_tests_properties(l${layout}_cross_${codec} PROPERTIES TIMEOUT 180 ENVIRONMENT "BTREE_CROSS_LEAF=1;BTREE_CROSS_CPU_TEST=1;BTREE_AGG_BACKPRESSURE=1;BTREE_BG_COMPACTION=1;BTREE_MEMORY_AUDIT=1;BTREE_SHARDS=8;BTREE_REUSE_UNCHANGED_BLOCKS=1")
    endforeach()
    add_test(NAME l${layout}_cross_fuzz COMMAND agg_compressed_crud_fuzz_l${layout})
    set_tests_properties(l${layout}_cross_fuzz PROPERTIES TIMEOUT 180 ENVIRONMENT "BTREE_CROSS_LEAF=1;BTREE_CROSS_CPU_TEST=1;BTREE_AGG_BACKPRESSURE=1;BTREE_BG_COMPACTION=1;BTREE_MEMORY_AUDIT=1;BTREE_SHARDS=8;BTREE_REUSE_UNCHANGED_BLOCKS=1")
    add_test(NAME l${layout}_cross_qpl_hardware COMMAND agg_check_l${layout} qpl)
    set_tests_properties(l${layout}_cross_qpl_hardware PROPERTIES TIMEOUT 180 SKIP_RETURN_CODE 77 ENVIRONMENT "BTREE_CROSS_LEAF=1;BTREE_CROSS_BYTES=16777216;BTREE_AGG_BACKPRESSURE=1;BTREE_BG_COMPACTION=1;BTREE_MEMORY_AUDIT=1;BTREE_SHARDS=8;BTREE_REUSE_UNCHANGED_BLOCKS=1;BTREE_QPL_COMPRESS_PATH=hardware;BTREE_QPL_GET_PATH=hardware;BTREE_QPL_MAINTENANCE_PATH=hardware")
    add_test(NAME l${layout}_async_metrics COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/check_async_metrics.py $<TARGET_FILE:agg_bench_l${layout}>)
    set_tests_properties(l${layout}_async_metrics PROPERTIES TIMEOUT 120)
  endif()
  if(layout GREATER 0)
    add_executable(agg_rebuild_equivalence_l${layout} rebuild_equivalence.c)
    target_link_libraries(agg_rebuild_equivalence_l${layout} agg_tree_l${layout})
    add_test(NAME l${layout}_rebuild_equivalence COMMAND agg_rebuild_equivalence_l${layout})
    set_tests_properties(l${layout}_rebuild_equivalence PROPERTIES TIMEOUT 120 ENVIRONMENT "BTREE_MEMORY_AUDIT=1")
    add_test(NAME l${layout}_split_measurement COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/check_split_measurement.py $<TARGET_FILE:agg_bench_l${layout}>)
    set_tests_properties(l${layout}_split_measurement PROPERTIES TIMEOUT 60)
    foreach(suite split_contract split_routes)
      add_executable(agg_${suite}_l${layout} ${suite}.c)
      target_link_libraries(agg_${suite}_l${layout} agg_tree_l${layout})
      add_test(NAME l${layout}_${suite} COMMAND agg_${suite}_l${layout})
      set_tests_properties(l${layout}_${suite} PROPERTIES TIMEOUT 60
        ENVIRONMENT "BTREE_MEMORY_AUDIT=1;BTREE_BG_COMPACTION=1;BTREE_SHARDS=1")
    endforeach()
    foreach(path software hardware)
      add_test(NAME l${layout}_qpl_routes_${path} COMMAND agg_split_routes_l${layout} ${path})
      set_tests_properties(l${layout}_qpl_routes_${path} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120
        ENVIRONMENT "BTREE_BG_COMPACTION=1;BTREE_MEMORY_AUDIT=1;BTREE_SHARDS=8")
    endforeach()
  endif()
  add_test(NAME l${layout}_trace COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/check_trace.py $<TARGET_FILE:agg_bench_l${layout}>)
  set_tests_properties(l${layout}_trace PROPERTIES TIMEOUT 120)
  foreach(suite history failures backpressure wait_protocol)
    add_executable(agg_${suite}_l${layout} ${suite}.c)
    target_link_libraries(agg_${suite}_l${layout} agg_tree_l${layout})
    add_test(NAME l${layout}_${suite} COMMAND agg_${suite}_l${layout})
    set_tests_properties(l${layout}_${suite} PROPERTIES TIMEOUT 120
      ENVIRONMENT "BTREE_BG_COMPACTION=1;BTREE_MEMORY_AUDIT=1;BTREE_SHARDS=1")
  endforeach()
  foreach(suite compressed_crud_fuzz compressed_failure_paths compressed_async_compaction)
    add_executable(agg_${suite}_l${layout} ../btree/${suite}.c)
    target_link_libraries(agg_${suite}_l${layout} agg_tree_l${layout})
    add_test(NAME l${layout}_${suite} COMMAND agg_${suite}_l${layout})
    set_tests_properties(l${layout}_${suite} PROPERTIES TIMEOUT 120
      ENVIRONMENT "BTREE_MEMORY_AUDIT=1;BTREE_FUZZ_SEED=11;BTREE_BG_COMPACTION=1")
  endforeach()
  foreach(codec raw lz4 zstd zlib)
    foreach(bg 0 1)
      add_test(NAME l${layout}_${codec}_bg${bg} COMMAND agg_check_l${layout} ${codec})
      set_tests_properties(l${layout}_${codec}_bg${bg} PROPERTIES TIMEOUT 120
        ENVIRONMENT "BTREE_BG_COMPACTION=${bg};BTREE_MEMORY_AUDIT=1;BTREE_SHARDS=1")
    endforeach()
  endforeach()
endforeach()
add_executable(agg_qpl_format qpl_format.c)
add_executable(agg_qpl_resources qpl_resources.c)
target_link_libraries(agg_qpl_resources agg_tree_l0)
add_test(NAME qpl_resources COMMAND agg_qpl_resources software 4)
set_tests_properties(qpl_resources PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120)
foreach(subpages 2 4)
  foreach(bg 0 1)
    add_test(NAME l0_legacy_subpages_${subpages}_bg${bg} COMMAND agg_check_l0 lz4 ${subpages})
    set_tests_properties(l0_legacy_subpages_${subpages}_bg${bg} PROPERTIES TIMEOUT 120
      ENVIRONMENT "BTREE_BG_COMPACTION=${bg};BTREE_MEMORY_AUDIT=1;BTREE_SHARDS=1")
  endforeach()
endforeach()
foreach(layout 2 3)
  foreach(seed 29 47)
    add_test(NAME l${layout}_fuzz_seed${seed} COMMAND agg_compressed_crud_fuzz_l${layout})
    set_tests_properties(l${layout}_fuzz_seed${seed} PROPERTIES TIMEOUT 120
      ENVIRONMENT "BTREE_BG_COMPACTION=1;BTREE_MEMORY_AUDIT=1;BTREE_SHARDS=8;BTREE_FUZZ_SEED=${seed};BTREE_BG_INCLUDE_TRIGGER=1")
  endforeach()
  add_test(NAME l${layout}_include_trigger COMMAND agg_check_l${layout} lz4)
  set_tests_properties(l${layout}_include_trigger PROPERTIES TIMEOUT 120
    ENVIRONMENT "BTREE_BG_COMPACTION=1;BTREE_MEMORY_AUDIT=1;BTREE_SHARDS=8;BTREE_BG_INCLUDE_TRIGGER=1")
endforeach()
target_link_libraries(agg_qpl_format agg_tree_l0)
add_test(NAME qpl_format COMMAND agg_qpl_format)
set_tests_properties(qpl_format PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120)
add_test(NAME intel_hybrid_runner COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/check_intel_hybrid.py $<TARGET_FILE:agg_bench_l3>)
set_tests_properties(intel_hybrid_runner PROPERTIES TIMEOUT 180)
